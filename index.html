import { createWorkflow } from "@mastra/core/workflows";

export const dailyPack = createWorkflow({
  id: "daily-content-pack",
  inputSchema: z.object({ date: z.string() }),
})
  .then(research)
  .foreach(writeForPlatform, { concurrency: 3 })
  .then(lint)
  .then(budgetGate)         // fails closed
  .then(requestApproval)    // suspends until Slack
  .commit();
